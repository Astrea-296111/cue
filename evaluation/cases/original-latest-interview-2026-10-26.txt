import re
import sqlite3
from typing import Any, Dict, List, Optional

class NL2SQLAgent:
    def __init__(self, db_path: str = ":memory:"):
        self.conn = sqlite3.connect(db_path)
        self.conn.row_factory = sqlite3.Row
        self.schema_text = """
CREATE TABLE item (item_id BIGINT PRIMARY KEY, item_title VARCHAR, category_id INT, seller_id BIGINT, price DECIMAL, status TINYINT);
CREATE TABLE trade_order (order_id BIGINT PRIMARY KEY, buyer_id BIGINT, seller_id BIGINT, item_id BIGINT, pay_amount DECIMAL, quantity INT, order_status TINYINT, pay_time DATETIME);
CREATE TABLE seller (seller_id BIGINT PRIMARY KEY, seller_name VARCHAR, seller_level TINYINT, category_id INT);
CREATE TABLE activity_signup (id BIGINT PRIMARY KEY, activity_id BIGINT, activity_name VARCHAR, item_id BIGINT, seller_id BIGINT, signup_status TINYINT, subsidy_price DECIMAL);
CREATE TABLE category (category_id INT PRIMARY KEY, category_name VARCHAR, parent_id INT, level TINYINT);
"""
        self.terms = {
            "GMV": "SUM(pay_amount) AND order_status IN (2,3,4)",
            "动销率": "有成交商品数 / 在售商品总数",
            "客单价": "GMV / 去重买家数",
            "百补商品": "activity_name LIKE '%百亿补贴%'",
            "报名通过率": "signup_status=2 数量 / 总报名数",
            "活跃商家": "近 30 天有成交的商家",
        }

    def _build_prompt(self, question: str) -> str:
        allowed_tables = ["item", "trade_order", "seller", "activity_signup", "category"]
        term_text = "\n".join([f"- {k}: {v}" for k, v in self.terms.items()])
        return f"""你是NL2SQL引擎。只允许使用以下表名：{allowed_tables}
Schema:
{self.schema_text}
业务术语:
{term_text}
请将用户问题转换为只读SELECT语句，只输出JSON：{{"sql": "..."}}
用户问题：{question}
"""

    def _safe_execute(self, sql: str) -> List[Dict[str, Any]]:
        clean = sql.strip().rstrip(";")
        if not re.match(r"^(SELECT|WITH)\b", clean, re.I):
            raise ValueError("仅允许只读 SELECT 查询")
        if re.search(r";\s*[A-Za-z]", clean):
            raise ValueError("不允许执行多条语句")
        forbidden = {"INSERT", "UPDATE", "DELETE", "DROP", "ALTER", "TRUNCATE", "CREATE", "ATTACH", "DETACH", "PRAGMA"}
        if any(re.search(rf"\b{kw}\b", clean, re.I) for kw in forbidden):
            raise ValueError("检测到禁止的写操作或系统命令")
        self.conn.execute("PRAGMA query_only = ON;")
        cur = self.conn.execute(clean + " LIMIT 1000")
        return [dict(r) for r in cur.fetchall()]

    def answer(self, question: str, llm_call: Optional[Any] = None) -> Dict[str, Any]:
        # 实际注入模型；测试时由 llm_call 返回 {"sql": "..."}
        if llm_call is None:
            raise RuntimeError("请注入 LLM 调用函数")
        import json
        raw = llm_call(self._build_prompt(question))
        sql = json.loads(raw)["sql"]
        rows = self._safe_execute(sql)
        return {"sql": sql, "rows": rows}

    def run_case(self) -> None:
        ddl = [
            "CREATE TABLE item (item_id BIGINT PRIMARY KEY, item_title VARCHAR, category_id INT, seller_id BIGINT, price DECIMAL, status TINYINT)",
            "CREATE TABLE trade_order (order_id BIGINT PRIMARY KEY, buyer_id BIGINT, seller_id BIGINT, item_id BIGINT, pay_amount DECIMAL, quantity INT, order_status TINYINT, pay_time DATETIME)",
            "CREATE TABLE seller (seller_id BIGINT PRIMARY KEY, seller_name VARCHAR, seller_level TINYINT, category_id INT)",
            "CREATE TABLE activity_signup (id BIGINT PRIMARY KEY, activity_id BIGINT, activity_name VARCHAR, item_id BIGINT, seller_id BIGINT, signup_status TINYINT, subsidy_price DECIMAL)",
            "CREATE TABLE category (category_id INT PRIMARY KEY, category_name VARCHAR, parent_id INT, level TINYINT)",
        ]
        for d in ddl:
            self.conn.execute(d)
        self.conn.executemany(
            "INSERT INTO trade_order VALUES (?, ?, ?, ?, ?, ?, ?, ?)",
            [
                (1, 101, 201, 301, 99.5, 1, 2, "2024-06-01 10:00:00"),
                (2, 102, 202, 302, 30.0, 2, 3, "2024-06-02 11:00:00"),
                (3, 101, 201, 303, 20.0, 1, 5, "2024-06-03 12:00:00"),
                (4, 103, 203, 304, 50.0, 1, 4, "2024-06-20 12:00:00"),
            ],
        )
        sql = ("SELECT SUM(pay_amount) AS gmv FROM trade_order "
               "WHERE order_status IN (2,3,4)")
        rows = self._safe_execute(sql)
        assert rows[0]["gmv"] == 179.5
        print("GMV 结果:", rows[0])

if __name__ == "__main__":
    agent = NL2SQLAgent()
    agent.run_case()
