```sql
SELECT
  SUM(CASE WHEN status IN (2,3,4) THEN pay_amount ELSE 0 END) AS gmv,
  COUNT(DISTINCT CASE WHEN status IN (2,3,4) THEN buyer_id END) AS buyers,
  ROUND(
    SUM(CASE WHEN status IN (2,3,4) THEN pay_amount ELSE 0 END) * 1.0
    / NULLIF(COUNT(DISTINCT CASE WHEN status IN (2,3,4) THEN buyer_id END), 0),
    2
  ) AS avg_order_value
FROM orders;
```
